# MVP v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Control-plane hexagonal (Python/FastAPI) que enrola máquinas, registra su presencia por heartbeat, las lista en un dashboard y permite conectar 1-click vía RustDesk, sin tocar el motor RustDesk.

**Architecture:** RustDesk OSS (ya dockerizado en `infra/`) es el motor A/V y no se consulta. El control-plane es la única fuente de verdad: los agentes se auto-registran (`POST /agents`) con un `enroll_token` de uso único y reportan presencia (`POST /agents/{id}/ping`). Dominio y casos de uso puros detrás de ports (Protocols); adapters FastAPI + SQLAlchemy/Postgres. Front estático consume la API.

**Tech Stack:** Python 3.12, FastAPI, Uvicorn, SQLAlchemy 2.x (Postgres), Pydantic v2, pytest + httpx TestClient. Agente: bash + curl + systemd timer. Front: HTML/JS estático (fetch).

## Global Constraints

- Python `>=3.12`. Hexagonal estricto: `domain/` y `application/` no importan `fastapi`, `sqlalchemy` ni nada de infra.
- RustDesk NO se toca ni se consulta (ni su sqlite). Registry = DB propia del control-plane.
- Tests EN repo + gate obligatorio + CI (política sme-cyber post PR#18).
- Port de persistencia: `DeviceRepository` con `save(device)`, `get(id)`, `list()`.
- Presencia: `Device.is_online(now, threshold_s=90)` = `last_seen` a menos de 90s.
- `enroll_token`: uso único, TTL 15 min, emitido por dashboard autenticado.
- Todos los endpoints de agente (`/agents`, `/agents/{id}/ping`) y de dashboard van autenticados (bearer token).

```python
# control-plane/src/domain/device.py
from dataclasses import dataclass, replace
from datetime import datetime

@dataclass(frozen=True)
class Device:
    id: str
    alias: str | None
    pubkey: str
    last_seen: datetime | None
    password: str

    def is_online(self, now: datetime, threshold_s: int = 90) -> bool:
        if self.last_seen is None:
            return False
        return (now - self.last_seen).total_seconds() < threshold_s

    def seen_at(self, moment: datetime) -> "Device":
        return replace(self, last_seen=moment)
```

```python
class DeviceRepository(Protocol):
    def save(self, device: Device) -> None: ...
    def get(self, device_id: str) -> Device | None: ...
    def list(self) -> list[Device]: ...

class EnrollTokenRepository(Protocol):
    def add(self, token: EnrollToken) -> None: ...
    def get(self, token: str) -> EnrollToken | None: ...
    def mark_used(self, token: str) -> None: ...
```

```python
# control-plane/src/adapters/memory_repo.py
from dataclasses import replace
from src.domain.device import Device
from src.domain.enrollment import EnrollToken

class InMemoryDeviceRepository:
    def __init__(self) -> None:
        self._data: dict[str, Device] = {}
    def save(self, device: Device) -> None:
        self._data[device.id] = device
    def get(self, device_id: str) -> Device | None:
        return self._data.get(device_id)
    def list(self) -> list[Device]:
        return list(self._data.values())

class InMemoryEnrollTokenRepository:
    def __init__(self) -> None:
        self._data: dict[str, EnrollToken] = {}
    def add(self, token: EnrollToken) -> None:
        self._data[token.token] = token
    def get(self, token: str) -> EnrollToken | None:
        return self._data.get(token)
    def mark_used(self, token: str) -> None:
        t = self._data.get(token)
        if t is not None:
            self._data[token] = replace(t, used=True)
```

```python
# control-plane/src/application/enroll.py
from datetime import datetime
from src.domain.device import Device
from src.application.ports import DeviceRepository, EnrollTokenRepository

class EnrollError(Exception):
    pass

def enroll_device(
    devices: DeviceRepository,
    tokens: EnrollTokenRepository,
    *,
    token: str,
    device_id: str,
    pubkey: str,
    alias: str | None,
    password: str,
    now: datetime,
) -> Device:
    et = tokens.get(token)
    if et is None or not et.is_valid(now):
        raise EnrollError("invalid or expired enroll token")
    tokens.mark_used(token)
    device = Device(id=device_id, alias=alias, pubkey=pubkey, last_seen=now, password=password)
    devices.save(device)
    return device
```

```python
# control-plane/src/adapters/sql_repo.py
from sqlalchemy import select, insert, update
from sqlalchemy.orm import Session
from src.adapters.db import devices_table, enroll_tokens_table
from src.domain.device import Device
from src.domain.enrollment import EnrollToken

class SqlDeviceRepository:
    def __init__(self, session: Session) -> None:
        self._s = session
    def save(self, device: Device) -> None:
        exists = self._s.execute(
            select(devices_table.c.id).where(devices_table.c.id == device.id)
        ).first()
        vals = dict(id=device.id, alias=device.alias, pubkey=device.pubkey,
                    last_seen=device.last_seen, password=device.password)
        if exists:
            self._s.execute(update(devices_table).where(devices_table.c.id == device.id).values(**vals))
        else:
            self._s.execute(insert(devices_table).values(**vals))
        self._s.commit()
    def get(self, device_id: str) -> Device | None:
        row = self._s.execute(select(devices_table).where(devices_table.c.id == device_id)).first()
        return self._to_device(row) if row else None
    def list(self) -> list[Device]:
        rows = self._s.execute(select(devices_table)).all()
        return [self._to_device(r) for r in rows]
    @staticmethod
    def _to_device(row) -> Device:
        m = row._mapping
        return Device(id=m["id"], alias=m["alias"], pubkey=m["pubkey"],
                      last_seen=m["last_seen"], password=m["password"])

class SqlEnrollTokenRepository:
    def __init__(self, session: Session) -> None:
        self._s = session
    def add(self, token: EnrollToken) -> None:
        self._s.execute(insert(enroll_tokens_table).values(
            token=token.token, created_at=token.created_at, ttl_s=token.ttl_s, used=token.used))
        self._s.commit()
    def get(self, token: str) -> EnrollToken | None:
        row = self._s.execute(select(enroll_tokens_table).where(enroll_tokens_table.c.token == token)).first()
        if not row:
            return None
        m = row._mapping
        return EnrollToken(token=m["token"], created_at=m["created_at"], ttl_s=m["ttl_s"], used=m["used"])
    def mark_used(self, token: str) -> None:
        self._s.execute(update(enroll_tokens_table).where(enroll_tokens_table.c.token == token).values(used=True))
        self._s.commit()
```

```python
# control-plane/src/adapters/web_api.py
import secrets
from datetime import datetime, timezone
from fastapi import FastAPI, Depends, HTTPException
from fastapi.staticfiles import StaticFiles
from pydantic import BaseModel
from src.adapters.auth import require_token
from src.adapters.settings import settings
from src.adapters.db import make_session_factory
from src.adapters.sql_repo import SqlDeviceRepository, SqlEnrollTokenRepository
from src.domain.enrollment import EnrollToken
from src.application.enroll import enroll_device, EnrollError
from src.application.heartbeat import record_heartbeat, UnknownDevice
from src.application.queries import list_devices

app = FastAPI(title="RemoteDesk control-plane")

_SessionFactory = None
def _session():
    global _SessionFactory
    if _SessionFactory is None:
        _SessionFactory = make_session_factory(settings.database_url)
    return _SessionFactory()

def get_device_repo() -> SqlDeviceRepository:
    return SqlDeviceRepository(_session())

def get_token_repo() -> SqlEnrollTokenRepository:
    return SqlEnrollTokenRepository(_session())

def _now() -> datetime:
    return datetime.now(timezone.utc).replace(tzinfo=None)

class EnrollBody(BaseModel):
    token: str
    id: str
    pubkey: str
    alias: str | None = None
    password: str

@app.post("/enroll-tokens", dependencies=[Depends(require_token)])
def create_enroll_token(tokens=Depends(get_token_repo)):
    et = EnrollToken(token=secrets.token_urlsafe(16), created_at=_now())
    tokens.add(et)
    return {"token": et.token}

@app.post("/agents", dependencies=[Depends(require_token)])
def enroll(body: EnrollBody, devices=Depends(get_device_repo), tokens=Depends(get_token_repo)):
    try:
        d = enroll_device(devices, tokens, token=body.token, device_id=body.id,
                          pubkey=body.pubkey, alias=body.alias, password=body.password, now=_now())
    except EnrollError as e:
        raise HTTPException(status_code=400, detail=str(e))
    return {"id": d.id, "alias": d.alias}

@app.post("/agents/{device_id}/ping", dependencies=[Depends(require_token)])
def ping(device_id: str, devices=Depends(get_device_repo)):
    try:
        record_heartbeat(devices, device_id=device_id, now=_now())
    except UnknownDevice:
        raise HTTPException(status_code=404, detail="unknown device")
    return {"ok": True}

@app.get("/agents", dependencies=[Depends(require_token)])
def agents(devices=Depends(get_device_repo)):
    return list_devices(devices, _now())
```

```bash
#!/usr/bin/env bash
# agent/install.sh — enrola esta máquina en el control-plane RemoteDesk.
# Uso: CP_URL=http://host:8000 API_TOKEN=... ENROLL_TOKEN=... RD_SERVER=host RD_KEY=... ./install.sh [alias]
set -euo pipefail

: "${CP_URL:?falta CP_URL}"; : "${API_TOKEN:?falta API_TOKEN}"
: "${ENROLL_TOKEN:?falta ENROLL_TOKEN}"; : "${RD_SERVER:?falta RD_SERVER}"; : "${RD_KEY:?falta RD_KEY}"
ALIAS="${1:-$(hostname)}"

command -v rustdesk >/dev/null || { echo "instalá el cliente rustdesk primero"; exit 1; }

# password permanente para acceso desatendido
PASSWORD="$(head -c 12 /dev/urandom | base64 | tr -dc 'A-Za-z0-9' | head -c 12)"

# configurar cliente rustdesk: server self-hosteado + key + password
rustdesk --config "rustdesk://config?relay=$RD_SERVER&api=$RD_SERVER&key=$RD_KEY" || true
rustdesk --password "$PASSWORD"

# esperar a que rustdesk genere el ID y leerlo
sleep 3
RD_ID="$(rustdesk --get-id 2>/dev/null | tr -dc '0-9')"
[ -n "$RD_ID" ] || { echo "no pude leer el RustDesk ID"; exit 1; }

# pubkey de la máquina (informativa para el registry)
PUBKEY="$(cat ~/.config/rustdesk/*.pub 2>/dev/null | head -1 || echo 'na')"

# auto-registro
curl -fsS -X POST "$CP_URL/agents" \
  -H "Authorization: Bearer $API_TOKEN" -H "Content-Type: application/json" \
  -d "{\"token\":\"$ENROLL_TOKEN\",\"id\":\"$RD_ID\",\"pubkey\":\"$PUBKEY\",\"alias\":\"$ALIAS\",\"password\":\"$PASSWORD\"}"

# guardar env para el heartbeat
sudo mkdir -p /etc/remotedesk
printf 'CP_URL=%s\nAPI_TOKEN=%s\nRD_ID=%s\n' "$CP_URL" "$API_TOKEN" "$RD_ID" | sudo tee /etc/remotedesk/agent.env >/dev/null
sudo cp heartbeat.sh /usr/local/bin/remotedesk-heartbeat.sh
sudo chmod +x /usr/local/bin/remotedesk-heartbeat.sh
sudo cp remotedesk-heartbeat.service remotedesk-heartbeat.timer /etc/systemd/system/
sudo systemctl daemon-reload
sudo systemctl enable --now remotedesk-heartbeat.timer
echo "OK: $ALIAS enrolada como RustDesk ID $RD_ID"
```
